1. Создать "Модель представления страницы"
2. Родительским объектом указать BaseModulePageV2
```JS
define("UsrOpportunityForecastHierarchyLevelPage", [],
	   function() {
	return {
		entitySchemaName: "UsrOpportunityForecastHierarchyLevel",
		methods: {},
		details: /**SCHEMA_DETAILS*/{}/**SCHEMA_DETAILS*/,
		diff: /**SCHEMA_DIFF*/[]/**SCHEMA_DIFF*/
	};
});
```

3. Запустить скрипт, создающий связь между объектом и страницей редактирования
```SQL
do
$body$
	declare
		"EntityName" text = 'UsrOpportunityForecastHierarchyLevel';
		"PageName" text = 'UsrOpportunityForecastHierarchyLevelPage';
		"PageCaption" text = 'Карточка схемы: "Иерархия уровней плана продаж"';
		
		"EntityId" uuid = (
			select sme."Id"
			from "SysModuleEntity" sme
			left outer join "SysSchema" ss on ss."UId" = sme."SysEntitySchemaUId"
			where ss."Name" = "EntityName"
		);
		
		"CardUId" uuid = (
			select "UId"
			from "SysSchema" ss
			where ss."Name" = "PageName"
		);
	begin
	
		insert into "SysModuleEdit"
		("CreatedById", "ModifiedById", "SysModuleEntityId", "UseModuleDetails", "CardSchemaUId", "PageCaption")
		values
		('410006e1-ca4e-4502-a9ec-e54d922d2c00', '410006e1-ca4e-4502-a9ec-e54d922d2c00', "EntityId", true, "CardUId", "PageCaption");
	end
$body$;
```
