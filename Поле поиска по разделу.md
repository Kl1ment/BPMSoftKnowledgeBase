attributes:{
	SmCustomSearchColumns: {

methods: {
	getFilters: function() {
		// Получаем фильтры
		var filters = this.callParent(arguments);
		// Получаем колонки, по которым будет производиться поиск
		var columns = this.get("SmCustomSearchColumns");
		// Получем строку поиска
		var text = this.get("SmCustomSearchText");
		if (columns.length > 0 && text){
			var filterGroup = this.BPMSoft.createFilterGroup();
			// Добавляем логический оператор OR для колонок
			filterGroup.logicalOperation = BPMSoft.LogicalOperatorType.OR;
			// Добавляем колонки в группу
			columns.forEach(column => {
				filterGroup.addItem(this.BPMSoft.createColumnFilterWithParameter(
					BPMSoft.ComparisonType.CONTAIN, column, text));
			});
			filters.add("CustomSearchFilter", filterGroup);
			this.set("SmCustomSearchText", "");
		}
		return filters;
	}
}
