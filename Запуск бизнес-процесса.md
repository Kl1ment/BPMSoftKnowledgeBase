private void GiveRights(string instanceName, Dictionary<string, string> baseParams, string entityId, bool giveRights = true) 
{ 
    var manager = _userConnection.ProcessSchemaManager;
    var processSchema = manager.GetInstanceByName(instanceName);
    var flowEngine = _systemUserConnection.ProcessEngine.ProcessExecutor;
    var param = new Dictionary<string, string>(baseParams)
    { 
        ["PgkEntityId"] = entityId,
        ["PgkGiveRights"] = giveRights.ToString() 
    };
    flowEngine.Execute(processSchema.UId, param); 
}
