define("Schemabeb62ef6Detail", [], // ← Схема может иметь другой код
    function() {
        return {
            entitySchemaName: "ScenarioSurvey", // ← Объект может иметь другой код
            methods: {
	            /**
	             * Добавляет корневую запись. Если не передан keepParentId, то Parent удаляется из DefaultValues, что делает запись «корневой»
	             */
                addRecord: function(editPageUId, keepParentId) {
                    if (!keepParentId) {
                        var defaultValues = this.get("DefaultValues");
                        var result = this.Ext.Array.filter(defaultValues, function(item) {
                            return (item.name !== "Parent"); // ← Должен быть код родительской колонки
                        }, this);
                        this.set("DefaultValues", result);
                    }
                    this.callParent(editPageUId);
                },
                
                /**
                 * Добавляет колонку Parent в набор отображаемых, если она отсутствует;
                 */
                getGridDataColumns: function() {
                    var gridDataColumns = this.callParent(arguments);
                    if (!gridDataColumns.Parent) {
                        gridDataColumns.Parent = {
                            path: "Parent" // ← Должен быть код родительской колонки ["Activity:Id:TmkRealRecordId"]
                        };
                    }
                    return gridDataColumns;
                },
                
                /**
                 * Добавляет свойство ParentId на основе поля Parent, чтобы DataGrid понимал иерархию
                 */
                prepareResponseCollection: function(collection) {
                    collection.each(function(item) {
                        var parent = item.get("Parent"); // ← Должен быть код родительской колонки
                        var parentId = parent && parent.value;
                        if (parentId) {
                            item.set("ParentId", parentId); // ← Должен быть код родительской колонки + Id
                        }
                        BPMSoft.each(item.columns, function(column) {
                            this.addColumnLink(item, column);
                            this.applyColumnDefaults(column);
                        }, this);
                    }, this);
                },
                
                /**
                 * Скрыты кнопки быстрого фильтра
                 */
                getHideQuickFilterButton: function() {
                    return false;
                },
                
                /**
                 * Скрыты кнопки быстрого фильтра
                 */
                getShowQuickFilterButton: function() {
                    return false;
                },

                updateDetail: function(config) {
                    config.reloadAll = true;
                    this.callParent([config]);
                },

                getAddChildElementButtonEnabled: function() {
                    return !this.Ext.isEmpty(this.getSelectedItems());
                },
                
                /**
                 * Добавляет дочернюю запись для выбранной строки. В DefaultValues явно указывается Parent — ID выбранной записи
                 */
                addChildElementRecord: function() {
                    var selectedItems = this.getSelectedItems();
                    if (this.Ext.isEmpty(selectedItems)) {
                        return;
                    }
                    var parentId = selectedItems[0];
                    var defaultValues = this.get("DefaultValues");
                    var result = this.Ext.Array.filter(defaultValues, function(item) {
                        return (item.name !== "Parent"); // ← Должен быть код родительской колонки
                    }, this);
                    result.push({
                            name: "Parent", // ← Должен быть код родительской колонки
                            value: parentId
                        });
                    this.set("DefaultValues", result);
                    this.addRecord(null, true);
                },

                clearSelection: function() {
                    this.set("activeRow", null);
                    this.set("selectedRows", null);
                },
                
                /**
                 * При удалении записей создает пакет (BatchQuery) и вызывает рекурсивное удаление всех дочерних элементов
                 */
                onDeleteAccept: function() {
                    var selectedRows = this.getSelectedItems();
                    var batch = this.Ext.create("BPMSoft.BatchQuery");
                    BPMSoft.each(selectedRows, function(recordId) {
                        this.deleteItem(recordId, batch, this);
                    }, this);
                    if (batch.queries.length > 0) {
                        batch.execute(this.onDeleted, this);
                    }
                },

                onDeleted: function(response) {
                    if (response && response.success) {
                        this.clearSelection();
                    } else {
                        this.showConfirmationDialog(
                            this.get("Resources.Strings.OnDeleteError")); // ← Не забудьте добавить локализуемую строку
                    }
                },
                
                /**
                 * Удаляет элемент и всех его потомков
                 */
                deleteItem: function(recordId, batch, scope) {
                    var grid = scope.getGridData();
                    var toDelete = new BPMSoft.Collection();
                    grid.each(function(item) {
                        var parent = item.get("Parent");
                        if (parent && parent.value === recordId) {
                            toDelete.add(item);
                        }
                    }, grid);
                    BPMSoft.each(toDelete.getItems(), function(item) {
                        this.deleteItem(item.get("Id"), batch, this);
                    }, scope);
                    if (grid.find(recordId)) {
                        var selfDelete = grid.get(recordId);
                        grid.remove(selfDelete);
                        var query = this.Ext.create("BPMSoft.DeleteQuery", {
                            rootSchema: scope.entitySchema
                        });
                        var filter = BPMSoft.createColumnFilterWithParameter(
                            BPMSoft.ComparisonType.EQUAL, "Id", recordId);
                        query.filters.addItem(filter);
                        batch.add(query);
                    }
                },
                
                /**
                 * Дополнительно проверяется право редактирования родительской записи (CanEditMasterRecord)
                 */
                getToolsVisible: function() {
                    return this.callParent(arguments) && this.get("CanEditMasterRecord");
                }
            },

            diff: /**SCHEMA_DIFF*/[
                {
                    "operation": "merge",
                    "name": "DataGrid",
                    "values": {
                        "hierarchical": true,
                        "hierarchicalColumnName": "ParentId"
                    }
                },
                {
                    "operation": "merge",
                    "name": "AddRecordButton",                
                    "parentName": "Detail",
                    "propertyName": "tools",                  
                    "index": 1,
                    "values": {                   
                        "itemType": BPMSoft.ViewItemType.BUTTON,
                        "menu": [],
                        "visible": {"bindTo": "getAddRecordButtonVisible"}
                    }
                },
                {
                    "operation": "insert",
                    "name": "AddParentRecordButton",
                    "parentName": "AddRecordButton",
                    "propertyName": "menu",
                    "values": {
                        "caption": {"bindTo": "Resources.Strings.AddRootElementButtonCaption"}, // ← Не забудьте добавить локализуемую строку
                        "click": {"bindTo": "addRecord"}
                    }
                },
                {
                    "operation": "insert",
                    "name": "AddChildElementButton",
                    "parentName": "AddRecordButton",
                    "propertyName": "menu",
                    "values": {
                        "caption": {"bindTo": "Resources.Strings.AddChildElementButtonCaption"}, // ← Не забудьте добавить локализуемую строку
                        "click": {"bindTo": "addChildElementRecord"},
                        "enabled": {"bindTo": "getAddChildElementButtonEnabled"}
                    }
                },
                {
                    "operation": "remove",
                    "name": "FiltersContainer"
                }
            ]/**SCHEMA_DIFF*/
        };
    }
);
