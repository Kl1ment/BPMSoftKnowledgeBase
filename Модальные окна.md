### Диалоговое окно
```JS
_requestActionComment: function(buttonCaption, maxCommentSize, callback, scope) {
	var controls = {
		comment: {
			dataValueType: BPMSoft.DataValueType.TEXT,
			customConfig: {
				className: "BPMSoft.MemoEdit",
				height: "77px",
				maxlength: maxCommentSize
			}
		}
	};
	var config = {
		defaultButton: 0,
		style: {
			borderStyle: "ts-messagebox-border-style-blue visa-action",
			buttonStyle: "orange"
		}
	};
	var yesButton = {
		className: "BPMSoft.Button",
		caption: buttonCaption,
		returnCode: "yes"
	};
	BPMSoft.utils.inputBox(caption, function(result, arg) {
		if (result === yesButton.returnCode) {
			var comment = arg.comment.value;
			if (comment && maxCommentSize && comment.length > maxCommentSize) {
				comment = comment.substring(0, maxCommentSize);
			}
			callback.call(scope, comment);
		}
	}, [yesButton, "cancel"], this, controls, config
						  );
},
```

### Загрузить модуль
```JS
var sandbox = this.sandbox;
var config = {
	widthPixels: "1000",
	heightPixels: "300"
};
var moduleId = sandbox.id + "_" + moduleName;
var renderTo = ModalBox.show(config, function() {
	sandbox.unloadModule(moduleId, renderTo);
});
sandbox.loadModule(moduleName, {
	id: moduleId,
	renderTo: renderTo,
	parameters: {
		RecordId: record.Id,
		RecordName: record.Name,
		EntitySchemaName: entitySchemaName,
		Uri: record.onNameLinkClick.url
	}
});
```
