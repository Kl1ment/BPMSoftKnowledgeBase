### Заполнить поля родительской запись из детали
```JS
// Родительская страница
messages: {
	SetColumnsValues: {
		mode: BPMSoft.MessageMode.PTP,
		direction: BPMSoft.MessageDirectionType.SUBSCRIBE
	}
},
methods: {
	onEntityInitialized: function(){
		this.callParent(arguments);
		this.subscribeMessages();
	},
	
	subscribeMessages: function() {
		var detailId = this.getDetailId("DetailName");
		var sandbox = this.sandbox;
		sandbox.subscribe("SetColumnsValues", this.onSetColumnsValues, this, [detailId]);
	},
	
	onSetColumnsValues: function(config) {
		var columnsValues = config.ColumnsValues;
		var isSave = config.IsSave;
		for (var columnName in columnsValues) {
			this.set(columnName, columnsValues[columnName]);
		}
		
		if (isSave) {
			this.save({ silent: true });
		}
	},
}
```

```JS
/**
 * Проверка на наличие активных записей по компании.
 * @param {Guid} companyId
 * @returns
 */
checkCompanyHasActiveRecord: function (companyId) {
	const recordId = this.get("Id");
	const esq = Ext.create("BPMSoft.EntitySchemaQuery", {
		rootSchemaName: "UsrKssPlanVacationCompanySetting"
	});

	esq.addColumn("Id");
	esq.addColumn("UsrAccount");
	esq.addColumn("UsrIsActive");

	const filterGroup = new BPMSoft.FilterGroup();

	const companyFilter = BPMSoft.createColumnFilterWithParameter(
		BPMSoft.ComparisonType.EQUAL,
		"UsrAccount.Id",
		companyId
	);
	filterGroup.addItem(companyFilter);

	const isActiveFilter = BPMSoft.createColumnFilterWithParameter(
		BPMSoft.ComparisonType.EQUAL,
		"UsrIsActive",
		true
	);
	filterGroup.addItem(isActiveFilter);

	if (!Ext.isEmpty(recordId)) {
		const notCurrentRecordFilter = BPMSoft.createColumnFilterWithParameter(
			BPMSoft.ComparisonType.NOT_EQUAL,
			"Id",
			recordId
		);
		filterGroup.addItem(notCurrentRecordFilter);
	}

	esq.filters.addItem(filterGroup);

	return new Promise(resolve => {
		esq.getEntityCollection(function (response) {
			if (response.success) {
				resolve(response.collection.getCount() === 0);
			} else {
				resolve(true);
			}
		});
	});
},
```

### Получить все функции и свойства объекта
```JS
function getProperties(obj) {
	var functions = [];
	var properties = [];
	for (var prop of Object.getOwnPropertyNames(obj)) {
		if (typeof obj[prop] === 'function') {
			functions.push(prop);
		} else {
			properties.push(prop);
		}
	}
	if (Object.getPrototypeOf(obj)) {
		var parentProperties = getProperties(Object.getPrototypeOf(obj));
		functions = functions.concat(parentProperties.functions);
		properties = properties.concat(parentProperties.properties);
	}
	return {
		functions: functions,
		properties:properties
	}
}
```
