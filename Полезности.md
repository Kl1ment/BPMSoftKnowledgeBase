### Заполнить поля родительской запись из детали
```JS
// Родительская страница
messages: {
	SetColumnsValues: {
		mode: BPMSoft.MessageMode.PTP,
		direction: BPMSoft.MessageDirectionType.SUBSCRIBE
	}
},
methods: {
	onEntityInitialized: function(){
		this.callParent(arguments);
		this.subscribeMessages();
	},
	
	subscribeMessages: function() {
		var detailId = this.getDetailId("DetailName");
		var sandbox = this.sandbox;
		sandbox.subscribe("SetColumnsValues", this.onSetColumnsValues, this, [detailId]);
	},
	
	onSetColumnsValues: function(config) {
		var columnsValues = config.ColumnsValues;
		var isSave = config.IsSave;
		for (var columnName in columnsValues) {
			this.set(columnName, columnsValues[columnName]);
		}
		
		if (isSave) {
			this.save({ silent: true });
		}
	},
}
```

### Получить все функции и свойства объекта
```JS
function getProperties(obj) {
	var functions = [];
	var properties = [];
	for (var prop of Object.getOwnPropertyNames(obj)) {
		if (typeof obj[prop] === 'function') {
			functions.push(prop);
		} else {
			properties.push(prop);
		}
	}
	if (Object.getPrototypeOf(obj)) {
		var parentProperties = getProperties(Object.getPrototypeOf(obj));
		functions = functions.concat(parentProperties.functions);
		properties = properties.concat(parentProperties.properties);
	}
	return {
		functions: functions,
		properties:properties
	}
}
```

### Ссылки FK на таблицу
```SQL
select 
	ref_table.name foreign_table,
	ref_column.name foreign_column
from sysreferences sr
left outer join sysobjects ref_table on ref_table.id = sr.fkeyid
left outer join sysobjects source_table on source_table.id = sr.rkeyid
left outer join  syscolumns ref_column on sr.fkeyid = ref_column.id and sr.fkey1 = ref_column.colid
where source_table.name = 'Case'
order by foreign_table
```

```SQL
select 
	parent_object.name foreign_table,
	parent_column.name foreign_column,
	fk.delete_referential_action_desc 
from sys.foreign_key_columns AS fkc
left outer join sys.foreign_keys fk on fk.object_id = fkc.constraint_object_id 
left outer join sys.objects parent_object on parent_object.object_id = fkc.parent_object_id 
left outer join sys.columns parent_column on fkc.parent_object_id = parent_column.object_id AND fkc.parent_column_id = parent_column.column_id
left outer join sys.objects referenced_object on referenced_object.object_id = fkc.referenced_object_id 
where referenced_object.name = 'Case'
order by foreign_table
```

### Отследить изменения в dom
```js
observeMutation: function() {
	const targetNode = document.getElementById("some-id");
	
	const config = { attributes: true, childList: true, subtree: true };
	const callback = (mutationList, observer) => { 
		for (const mutation of mutationList) { 
			if (mutation.type === "childList") { 
				console.log("A child node has been added or removed."); 
			} else if (mutation.type === "attributes") { 
				console.log(`The ${mutation.attributeName} attribute was modified.`); 
			} 
		} 
	};
	const observer = new MutationObserver(callback);
	
	observer.observe(targetNode, config);
}
```
