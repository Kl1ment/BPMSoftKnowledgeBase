Переопределяем базовый метод
```JS
setValidationConfig: function() {
	this.callParent(arguments);
	this.addColumnValidator("UsrShiftAmount", this.shiftAmountValidator);
},
```

В методе валидации возвращаем объект с полем `invalidMessage`. Если сообщение пустое, поле прошло валидацию. Иначе будет выводиться ошибка

```JS
shiftAmountValidator: function(value) {
	var shiftAmount = value || this.$UsrShiftAmount;
	var message = {
		invalidMessage: ""
	};

	if (!(this.$UsrOpportunityShiftType.value == ClientConstants.UsrOpportunityShiftType.Shift ||
		  this.$UsrOpportunityShiftType.value == ClientConstants.UsrOpportunityShiftType.AmountDecrease)) {
		return message;
	}

	if (this.$UsrShiftAmount > this.$Remains){
		message.invalidMessage = "Количество не может быть больше остатка";
	}

	return message;
},
```
